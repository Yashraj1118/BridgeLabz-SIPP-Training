Class AttendanceSystem
    Attribute: attendance[students][days]
    Method markAttendance(student, day, status):
        attendance[student][day] = status
    Method updateAttendance(student, day, newStatus):
        attendance[student][day] = newStatus
    Method generateReport(student):
        count = count 'P' in attendance[student]
        Print "Student present for", count, "days."

Main:
    system = AttendanceSystem(30 students, 31 days)
    system.markAttendance(1, 1, "P")
    system.markAttendance(1, 2, "P")
    system.markAttendance(1, 3, "A")

    system.updateAttendance(1, 3, "P")
    system.generateReport(1)
